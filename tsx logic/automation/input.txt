import React from 'react';
import { useCurrentFrame, useVideoConfig, interpolate, Easing, spring, AbsoluteFill } from 'remotion';

export const compositionConfig = {
  id: 'Scene01_SlowGrind',
  durationInSeconds: 18.67,
  fps: 30,
  width: 1920,
  height: 1080,
};

const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

const Scene01_SlowGrind: React.FC = () => {
  const frame = useCurrentFrame();
  const { fps, durationInFrames } = useVideoConfig();

  const sp = (delay: number, config?: object) =>
    spring({ frame: Math.max(0, frame - delay), fps, config: { damping: 12, mass: 0.4, stiffness: 120, ...config } });

  const exitOp = interpolate(frame, [durationInFrames - 15, durationInFrames], [1, 0], {
    extrapolateLeft: 'clamp', extrapolateRight: 'clamp',
  });

  // camera slow push-in
  const camZoom = interpolate(frame, [0, durationInFrames], [1, 1.08], { extrapolateRight: 'clamp' });

  // grinding staircase bars (decline)
  const bars = [0, 1, 2, 3, 4, 5];

  return (
    <AbsoluteFill style={{ backgroundColor: '#7A3E3E', overflow: 'hidden', opacity: exitOp }}>
      {/* Gradient + Shape Layers + Texture + Particles + Content + Character + Vignette + Watermark */}

      {/* Dynamic gradient background */}
      <AbsoluteFill style={{
        background: 'radial-gradient(circle at 30% 20%, #9A5555 0%, #7A3E3E 45%, #4A2424 100%)',
      }} />

      {/* Punk collage torn shapes */}
      <AbsoluteFill style={{ transform: `scale(${camZoom})` }}>
        {[...Array(7)].map((_, i) => {
          const r = seededRandom(i + 3);
          const rot = interpolate(frame, [0, durationInFrames], [r * 40 - 20, r * 40]);
          return (
            <div key={i} style={{
              position: 'absolute',
              left: `${r * 90}%`,
              top: `${seededRandom(i + 9) * 85}%`,
              width: 140 + r * 220,
              height: 90 + seededRandom(i + 5) * 160,
              backgroundColor: i % 2 === 0 ? '#000000' : '#F4F2F2',
              opacity: 0.05,
              transform: `rotate(${rot}deg)`,
              clipPath: 'polygon(0 8%, 100% 0, 96% 100%, 4% 92%)',
            }} />
          );
        })}
      </AbsoluteFill>

      {/* Motion texture: diagonal grind lines */}
      <AbsoluteFill style={{ opacity: 0.08 }}>
        {[...Array(18)].map((_, i) => (
          <div key={i} style={{
            position: 'absolute',
            top: 0,
            left: `${(i / 18) * 100}%`,
            width: 2,
            height: '100%',
            backgroundColor: '#000000',
            transform: `skewX(-14deg) translateY(${Math.sin(frame * 0.02 + i) * 8}px)`,
          }} />
        ))}
      </AbsoluteFill>

      {/* Floating particles */}
      {[...Array(26)].map((_, i) => {
        const r = seededRandom(i + 1);
        const y = (seededRandom(i + 2) * 100 + frame * 0.15) % 100;
        return (
          <div key={i} style={{
            position: 'absolute',
            left: `${r * 100}%`,
            top: `${y}%`,
            width: 3 + r * 6,
            height: 3 + r * 6,
            borderRadius: '50%',
            backgroundColor: '#F4F2F2',
            opacity: 0.25,
          }} />
        );
      })}

      {/* Declining staircase bars (icon-graphic) */}
      <div style={{
        position: 'absolute',
        right: 140,
        bottom: 180,
        display: 'flex',
        alignItems: 'flex-end',
        gap: 14,
      }}>
        {bars.map((b) => {
          const grow = sp(30 + b * 8);

            <div key={b} style={{
              width: 42,
              height: h,
              backgroundColor: b > 3 ? '#E0A060' : '#F4F2F2',
              opacity: 0.9,
              borderRadius: 4,
            }} />
          );
        })}
      </div>

      {/* Main content */}
      <div style={{
        position: 'absolute',
        left: 130,
        top: 300,
        transform: `translateY(${(1 - sp(15)) * 60}px)`,
        opacity: sp(15),
      }}>
        <div style={{
          fontFamily: "'Noto Sans Devanagari'",

          fontWeight: 800,
          opacity: 0.55,
          marginBottom: 10,

        }}>
          मिडल क्लास<br />अचानक गरीब<br />नहीं बनता

        position: 'absolute',
