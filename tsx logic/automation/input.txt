const seededRandom = (seed: number): number => {
  const x = Math.sin(seed * 9999) * 10000;
  return x - Math.floor(x);
};

  const camRot = Math.sin(frame * 0.006) * 1;

    const t = frame - hit;
    if (t < 0 || t > 22) return 0;
    return Math.sin(t * 1.6) * (1 - t / 22) * 15;
  };

  const blindIn = sp(6);

    extrapolateRight: 'clamp',

    extrapolateRight: 'clamp',
