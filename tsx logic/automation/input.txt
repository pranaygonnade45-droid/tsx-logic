        left: -240, top: -240, borderRadius: '50%',
      }} />

      {/* Motion texture: flowing horizontal dashes */}
      {Array.from({ length: 9 }).map((_, i) => (
        <div key={`dash-${i}`} style={{
          position: 'absolute', top: 70 + i * 120,
          left: ((frame * (1 + i * 0.2)) % 2100) - 180,
          width: 90, height: 2, backgroundColor: '#EBEDE3', opacity: 0.06,
        }} />
      ))}

      {/* Floating particles */}
      {Array.from({ length: 16 }).map((_, i) => (
        <div key={`p-${i}`} style={{
          position: 'absolute',
          left: (seededRandom(i * 9) * 1920 + frame * (0.4 + seededRandom(i) * 0.5)) % 1920,
          top: seededRandom(i * 5) * 1080,
          width: 4, height: 4, borderRadius: '50%',
          backgroundColor: i % 3 === 0 ? '#FFF449' : '#EBEDE3', opacity: 0.12,
        }} />
      ))}

      {/* Main content */}
      <AbsoluteFill style={{ transform: `scale(${camScale}) translateX(${camX}px)` }}>
        {/* Headline */}
        <div style={{
          position: 'absolute', left: 0, right: 0, top: 90, textAlign: 'center',
          opacity: sp(5), transform: `translateY(${(1 - sp(5)) * -50}px)`,
        }}>
          <div style={{
            fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 88, fontWeight: 900,
            color: '#EBEDE3', letterSpacing: -1,
          }}>
            YOUR STRESS IS THEIR <span style={{ color: '#FFF449' }}>REVENUE</span>
          </div>
        </div>

        {/* Extraction flow diagram: YOU → channels → SYSTEM */}
        {/* YOU node */}
        <div style={{
          position: 'absolute', left: 170, top: 470, width: 240, height: 150,
          borderRadius: 26, border: '3px solid #EBEDE3', backgroundColor: 'rgba(15,33,29,0.8)',
          display: 'flex', alignItems: 'center', justifyContent: 'center',
          fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 52, fontWeight: 900, color: '#EBEDE3',
          opacity: sp(40), transform: `scale(${0.7 + sp(40) * 0.3})`,
        }}>YOU</div>

        {/* SYSTEM node */}
        <div style={{
          position: 'absolute', right: 150, top: 440, width: 320, height: 210,
          borderRadius: 26, border: '3px solid #FFF449', backgroundColor: 'rgba(15,33,29,0.85)',
          display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center',
          boxShadow: `0 0 ${28 + Math.sin(frame / 9) * 12}px rgba(255,244,73,0.22)`,
          opacity: sp(60), transform: `scale(${0.7 + sp(60) * 0.3})`,
        }}>
          <div style={{ fontFamily: 'Mont Heavy, Inter, sans-serif', fontSize: 46, fontWeight: 900, color: '#FFF449' }}>
            THE SYSTEM
          </div>
          <div style={{ fontFamily: 'Inter, sans-serif', fontSize: 24, color: '#EBEDE3', opacity: 0.75, marginTop: 8 }}>
            collects on every layer
          </div>
        </div>

        {/* Channel pipes with traveling payment dots */}
        {CHANNELS.map((c, i) => {
          const s = sp(90 + i * 22);
          const midY = c.y;
          return (
            <div key={c.label}>
              {/* pipe */}
              <svg style={{ position: 'absolute', left: 0, top: 0 }} width={1920} height={1080}>
                <path
                  d={`M 410 545 C 650 545, 650 ${midY} 900 ${midY} C 1150 ${midY}, 1200 545, 1450 545`}
                  fill="none" stroke="#EBEDE3" strokeOpacity={0.22 * s} strokeWidth={3} strokeDasharray="10 10"
                  strokeDashoffset={-frame * 1.2}
                />
              </svg>
              {/* channel label chip */}
              <div style={{
                position: 'absolute', left: 720, top: midY - 52, width: 380, textAlign: 'center',
                opacity: s, transform: `translateY(${(1 - s) * 40}px)`,
              }}>
                <div style={{
                  display: 'inline-block', padding: '12px 26px', borderRadius: 40,
                  backgroundColor: 'rgba(15,33,29,0.9)', border: '2px solid #FFF449',
                }}>
                  <div style={{ fontFamily: 'Inter, sans-serif', fontSize: 25, fontWeight: 800, color: '#EBEDE3' }}>{c.label}</div>
                  <div style={{ fontFamily: 'JetBrains Mono, monospace', fontSize: 22, color: '#FFF449', marginTop: 4 }}>{c.tag}</div>
                </div>
              </div>
              {/* traveling dot */}
              {frame > 150 + i * 20 && (() => {
                const t = ((frame - 150 - i * 20) % 90) / 90;
                const dx = interpolate(t, [0, 1], [420, 1440]);
                const arc = Math.sin(t * Math.PI);
                const dy = 545 + (midY - 545) * arc;
                return (
                  <div style={{
                    position: 'absolute', left: dx, top: dy - 7, width: 14, height: 14,
                    borderRadius: '50%', backgroundColor: '#FFF449',
                    boxShadow: '0 0 16px rgba(255,244,73,0.9)',
                    opacity: t < 0.05 || t > 0.95 ? 0 : 1,
                  }} />
                );
              })()}
            </div>
          );
        })}

        {/* Bottom takeaway */}
        <div style={{
          position: 'absolute', left: 0, right: 0, bottom: 90, textAlign: 'center',
          fontFamily: 'Inter, Poppins, sans-serif', fontSize: 42, fontWeight: 800, color: '#EBEDE3',
          opacity: sp(280), transform: `translateY(${(1 - sp(280)) * 50}px)`,
        }}>
          Even a "solution" can quietly become <span style={{ color: '#FFF449' }}>another expense.</span>
        </div>
      </AbsoluteFill>

      {/* Vignette overlay */}
      <AbsoluteFill style={{ background: 'radial-gradient(ellipse at center, transparent 50%, rgba(0,0,0,0.58) 100%)' }} />

      {/* Watermark */}
      <div style={{
        position: 'absolute', bottom: 30, right: 45, fontFamily: 'Inter, sans-serif',
        fontSize: 22, fontWeight: 600, color: '#9E9E9E', opacity: 0.3,
      }}>
        KuxAI.7x
      </div>
    </AbsoluteFill>
  );
};

export default Scene50StressIsRevenue;
